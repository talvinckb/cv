%% #1 domaine, #2 contexte (équipe), #3 prose, #4 pastilles.
%% varwidth compose les pastilles jusqu'à \projmaxtag mais rétrécit à la largeur
%% réellement occupée : chaque ligne garde donc sa propre gouttière.
\newcommand{\proj}[4]{%
  \sbox\projtagbox{\begin{varwidth}[t]{\projmaxtag}\raggedleft #4\end{varwidth}}%
  \setlength{\projtagw}{\wd\projtagbox}%
  \setlength{\projprosew}{\dimexpr\textwidth-\projlabelw-\projgapa-\projgapb-\projtagw\relax}%
  \noindent
  \parbox[t]{\projlabelw}{\raggedright\hyphenpenalty=10000\exhyphenpenalty=10000
    \textcolor{accent}{\textbf{#1}}\\[1.5pt]{\small\color{soft}#2}}%
  \hspace{\projgapa}%
  \parbox[t]{\projprosew}{\RaggedRight #3}%
  \hspace{\projgapb}%
  \usebox\projtagbox
  \par
}
